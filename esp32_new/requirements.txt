# ==================== 开发电脑上需要安装的工具 ====================

# 1. 固件烧录工具 (必需) - 用于烧录 MicroPython 固件到 ESP32
esptool>=3.0

# 2. 串口通信库 (必需) - esptool 的依赖，用于串口通信
pyserial>=3.5

# 3. 文件管理工具 (推荐) - 用于上传代码到 ESP32 (替代 Thonny)
mpremote>=0.4.0

# ==================== ESP32 MicroPython 内置库 (无需安装) ====================
#
# 以下库已内置在所有标准 MicroPython 固件中，无需安装
#
# 【硬件接口】
#   machine              - GPIO, I2C, PWM, Timer, UART, ADC 等
#   micropython          - MicroPython 内部优化函数
#
# 【数据结构和类型】
#   json                 - JSON 解析 (读写校准文件)
#   struct               - 二进制数据打包和解包
#   collections          - deque (数据缓冲)
#   array                - 数组对象
#
# 【时间和数学】
#   time                 - 时间管理、延迟、计时
#   math                 - 三角函数、数学运算
#   random               - 随机数生成
#
# 【可选库】(项目不使用，但 MicroPython 支持)
#   socket               - 网络通信
#   network              - WiFi 连接
#   ubluetooth           - 蓝牙支持
#   uos                  - 文件系统操作
#   usys                 - 系统参数
#   uasyncio             - 异步编程
